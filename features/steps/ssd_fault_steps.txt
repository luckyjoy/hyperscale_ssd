import os

from behave import given, when, then

SYSTEM = platform.system()
MAX_LATENCY_MS = 50
MIN_IOPS = 1000
MIN_THROUGHPUT_MB = 50

def run_cmd(cmd, shell=False):
    print(f"[+] Running: {cmd}")
    subprocess.run(cmd, shell=shell, check=True)

# ----------------- GIVEN -----------------
@given("a VM is launched with NVMe attached")
def step_launch_vm(context):
    """Launch a single NVMe SSD VM."""
    if SYSTEM == "Windows":
        run_cmd(["python", "qemu-faults\\qemu_launch_vm.py"])
    else:
        run_cmd(["python3", "qemu-faults/qemu_launch_vm.py"])
    time.sleep(3)

        run_cmd(["python3", "qemu-faults/qemu_launch_vm.py", "--devices", "2"])

@when("I inject 5 random latency spikes between 50-200ms")
def step_latency_spikes(context):
    for _ in range(5):
        delay = random.randint(50, 200)
        print(f"[+] Injecting latency {delay}ms")
        if SYSTEM == "Windows":
            run_cmd(["python", "qemu-faults\\nbd_delay.py", "10810", str(delay)])
        else:
            run_cmd(["./qemu-faults/nbdkit_delay_server.sh", "./vm-disk.qcow2", "10810", str(delay)])

# ----------------- THEN -----------------
@then("the VM should be running")
def step_vm_running(context):
    print(f"[+] VM running on {SYSTEM}")

@then("IO latency and performance should be within acceptable limits")
def step_verify_io_performance(context):
    fio_json_file = "reports\\fio_guest.json" if SYSTEM == "Windows" else "reports/fio_guest.json"
    if not os.path.exists(fio_json_file):
        raise FileNotFoundError(f"{fio_json_file} not found.")
    with open(fio_json_file, "r") as f:
        data = json.load(f)
    jobs = data.get("jobs", [])
    if not jobs:
        raise ValueError("No jobs found in fio output")
    failed = False
    for job in jobs:
        lat_avg = job['read']['lat_ns']['mean']/1e6
        lat_max = job['read']['lat_ns']['max']/1e6
        iops = job['read']['iops']
        bw_mb = job['read']['bw_bytes']/(1024*1024)
        print(f"[+] Job {job.get('jobname','unknown')}: Avg Latency={lat_avg:.2f}ms, Max={lat_max:.2f}ms, IOPS={iops}, BW={bw_mb:.2f}MB/s")
        if lat_avg > MAX_LATENCY_MS or iops < MIN_IOPS or bw_mb < MIN_THROUGHPUT_MB:
            failed = True
    if failed:
        raise AssertionError("IO performance verification failed.")
