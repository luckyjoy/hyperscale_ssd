import time
from behave import given, when, then
import subprocess
import json
import os
import re

class FIOExecutor:
    """
    A class to execute fio commands and parse the JSON output.
    """
    def __init__(self, fio_path="fio.exe"):
        """
        Initializes the FIOExecutor.
        :param fio_path: The path to the fio executable. Defaults to "fio.exe"
                         assuming it's in your system's PATH.
        """
        self.fio_path = fio_path
        self.results = {}

    def _execute_fio_command(self, job_name, rw_type, bs, size, iodepth, rwmixread=None, runtime_sec=30):
        """Builds and executes a fio command, returning parsed JSON results."""
        output_file = f"{job_name}_output.json"
        
        # Build the command using a list for subprocess
        command = [
            self.fio_path,
            '--name=' + job_name,
            '--rw=' + rw_type,
            f'--bs={bs}',
            f'--size={size}',
            f'--iodepth={iodepth}',
            '--direct=1',
            '--numjobs=1',
            f'--runtime={runtime_sec}',
            '--output-format=json',
            '--output=' + output_file,
            '--filename=test_file.dat'
        ]
        
        # Add rwmixread if specified
        if rwmixread is not None:
            command.append(f'--rwmixread={rwmixread}')
        
        try:
            print(f"Executing FIO command: {' '.join(command)}")
            subprocess.run(command, check=True, capture_output=True, text=True)
            
            with open(output_file, 'r') as f:
                json_data = json.load(f)
            
            os.remove(output_file) # Clean up the output file
            # os.remove('test_file.dat') # FIO removes this automatically after run
            
            return json_data
            
        except subprocess.CalledProcessError as e:
            print(f"Error executing fio: {e.stderr}")
            raise
        except FileNotFoundError:
            print(f"Error: fio executable not found at '{self.fio_path}'. Please check your PATH or provide the full path.")
            raise
        except Exception as e:
            print(f"An unexpected error occurred: {e}")
            raise

    def sequential_write(self, size_gb, bs_kb, runtime_sec):
        json_results = self._execute_fio_command(
            job_name='sequential_write',
            rw_type='write',
            bs=f'{bs_kb}k',
            size=f'{size_gb}g',
            iodepth=32,
            runtime_sec=runtime_sec
        )
        job_data = json_results['jobs'][0]
        self.results['write_throughput'] = job_data['write']['bw_mean'] / 1024 / 1024 # Convert B/s to GB/s
        self.results['write_latency'] = job_data['write']['lat_ns']['mean'] / 1000 # Convert ns to us

    def mixed_random(self, rwmix_str, bs_kb='4', iodepth=32, runtime_sec=30):
        read_percentage = int(rwmix_str.split('/')[0])
        json_results = self._execute_fio_command(
            job_name=f'mixed_random',
            rw_type='randrw',
            bs=f'{bs_kb}k',
            size='4g',
            iodepth=iodepth,
            rwmixread=read_percentage,
            runtime_sec=runtime_sec
        )
        job_data = json_results['jobs'][0]
        self.results['mixed_read_iops'] = job_data['read']['iops_mean']
        self.results['mixed_write_iops'] = job_data['write']['iops_mean']
        
@given('a mounted SSD volume')
def step_impl(context):
    """Sets up the test by initializing the FIOExecutor."""
    context.test_tool = FIOExecutor()

@given('the volume is pre-filled with data')
def step_impl(context):
    """
    Simulates a volume that is already in a used state. This step is a placeholder
    as it would be complex to implement in a single step.
    """
    print("Pre-filling the volume with data... (simulated)")

@when('I perform a sustained mixed random workload with a read/write mix of "{rwmix_str}"')
def step_impl(context, rwmix_str):
    """Executes a mixed random workload."""
    context.test_tool.mixed_random(rwmix_str)

@when('I perform a "{runtime:d}" hour sustained sequential write of "{size_gb:d}" GB with a block size of "{bs_kb:d}" KB')
def step_impl(context, runtime, size_gb, bs_kb):
    """Executes a sustained sequential write for a specified number of hours."""
    context.test_tool.sequential_write(size_gb, bs_kb, runtime * 3600)

@then('the average read IOPS should be at least "{expected_value:d}"')
def step_impl(context, expected_value):
    """Asserts that the measured read IOPS meets the required minimum."""
    actual_value = context.test_tool.results['mixed_read_iops']
    assert actual_value >= float(expected_value), f"Expected random read IOPS of {expected_value}, but got {actual_value:.2f}."

@then('the average write IOPS should be at least "{expected_value:d}"')
def step_impl(context, expected_value):
    """Asserts that the measured write IOPS meets the required minimum."""
    actual_value = context.test_tool.results['mixed_write_iops']
    assert actual_value >= float(expected_value), f"Expected write IOPS of {expected_value}, but got {actual_value:.2f}."

@then('the average write throughput should be at least "{expected_value:f}" GB/s')
def step_impl(context, expected_value):
    """Asserts that the measured throughput meets the required minimum."""
    actual_value = context.test_tool.results['write_throughput']
    assert actual_value >= float(expected_value), f"Expected write throughput of {expected_value:.2f} GB/s, but got {actual_value:.2f} GB/s."

@then('the average write latency should be less than "{expected_value:d}" microseconds')
def step_impl(context, expected_value):
    """Asserts that the measured latency is within acceptable limits."""
    actual_value = context.test_tool.results['write_latency']
    assert actual_value < float(expected_value), f"Expected write latency to be less than {expected_value} us, but was {actual_value:.2f} us."
