import random
import time
from behave import given, when, then

# ------------------------------------------------
# Long Duration FIO Workloads
# ------------------------------------------------


@when("I run a sustained {percent:d}% write FIO workload for {hours:d} hour")
def step_impl(context, percent, hours):
    key = f"{percent}w_{hours}h"
    baseline = 100000
    throughput = baseline - (100 - percent) * 500
    context.fio_results[key] = {"throughput": throughput, "errors": 0}

# ------------------------------------------------
# Read/Write Mix Workloads
# ------------------------------------------------

@when("I run a {read_pct:d}/{write_pct:d} read/write workload for {duration:d} minutes")
def step_impl(context, read_pct, write_pct, duration):
    key = f"{read_pct}r_{write_pct}w_{duration}m"
    throughput = random.randint(80000, 100000)  # simulate workload performance
    latency = random.uniform(1.0, 5.0)
    context.fio_results[key] = {"throughput": throughput, "latency": latency}

@then("SSD throughput should remain stable")
def step_impl(context):
    for test, result in context.fio_results.items():
        assert result["throughput"] > 0, f"{test} throughput dropped unexpectedly"

@then("latency should remain below {latency: d} ms")
def step_impl(context, latency):
    for test, result in context.fio_results.items():
        assert result["latency"] <= latency, f"{test} latency {result['latency']}ms exceeded {latency}ms"

# ------------------------------------------------
# Multi-VM Workloads
# ------------------------------------------------

@given("{vm_count:d} VMs are launched sharing the same NVMe SSD")
def step_impl(context, vm_count):
    # initial latency scales with number of VMs
    context.vms = [{"id": i, "latency": random.uniform(1.0, vm_count * 0.5)} for i in range(1, vm_count + 1)]

@when("each VM runs random {read_pct:d}/{write_pct:d} read/write workloads")
def step_impl(context, read_pct, write_pct):
    for vm in context.vms:
        vm["latency"] = random.uniform(1.0, 5.0)

@then("latency for all VMs should remain below {latency:d} ms average")
def step_impl(context, latency):
    for vm in context.vms:
        assert vm["latency"] <= latency, f"VM{vm['id']} latency {vm['latency']}ms exceeded {latency}ms"
